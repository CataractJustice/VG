}

MarchingCubesVoxelMesh::MarchingCubesVoxelMesh(VoxelData* data, float threshold, unsigned int levelofdetails, glm::vec3 offset) : vertices(0), normals(0), textureOffsets(0), indices(0)
{
	this->data = data;
	this->offset = offset;
	this->threshold = threshold;
	this->levelofdetails = levelofdetails;
	dimensions = data->getDimensions();
}

void MarchingCubesVoxelMesh::updateMesh()
{
	clear();
	const int verts = dimensions.x * dimensions.y * dimensions.z * 6;
	const int lineMass = dimensions.z;
	const int sliceMass = dimensions.y * dimensions.z;
	this->vertices.reserve(verts);
	this->normals.reserve(verts);
	this->textureOffsets.reserve(verts);
	this->indices.reserve(verts);

	for (int i = (data->isChunkEmpty() || data->isChunkFilled()) ? dimensions.x - 1 : 0; i < dimensions.x; i++)
	{
		bool skipSlice = false;
		if (i + 1 < dimensions.x) 
		{
			int m0 = data->getXSliceMass(i);
			int m1 = data->getXSliceMass(i + 1);

			if ((m0 == 0 && m1 == 0) || (m0 == sliceMass && m1 == sliceMass)) 
			{
				skipSlice = true;
			}
		}
		for (int j = skipSlice ? dimensions.y - 1 : 0; j < dimensions.y; j++)
		{
			bool skipLine = false;
			if ((i + 1 < dimensions.x) && (j + 1 < dimensions.y)) {
				int m00 = data->getXYLineMass(i, j);
				int m10 = data->getXYLineMass(i + 1, j);
				int m01 = data->getXYLineMass(i, j + 1);
				int m11 = data->getXYLineMass(i + 1, j + 1);

				if ((m00 == 0 && m01 == 0 && m10 == 0 && m11 == 0) || (m00 == lineMass && m01 == lineMass && m10 == lineMass && m11 == lineMass)) 
				{
					skipLine = true;
				}
			}
			for (int k = skipLine ? dimensions.z - 1 : 0; k < dimensions.z; k++)
			{
				Voxel* voxelAtIndex = data->getVoxelWrapped(glm::ivec3(i,j,k));
				const VoxelMaterial materialAtIndex = VoxelMaterials::getInstance().getById(voxelAtIndex->voxelTypeId);
				//CUBE START
				if(materialAtIndex.cube) 
				{
					if(voxelAtIndex->amount <= 0) continue;
					const glm::ivec3 neighbors[] = {
						glm::ivec3(-1, 0, 0),
						glm::ivec3(1, 0, 0),
						glm::ivec3(0, -1, 0),
						glm::ivec3(0, 1, 0),
						glm::ivec3(0, 0, -1),
						glm::ivec3(0, 0, 1),
					};
					for(int n = 0; n < 6; n++) 
					{
						Voxel* nv = data->getVoxelWrapped(glm::ivec3(i,j,k) + neighbors[n]);
						if(nv->amount > 0) 
						{
							VoxelMaterial mat = VoxelMaterials::getInstance().getById(nv->voxelTypeId);
							if(mat.cube) continue;
						}
						int addedVerts[] = 
						{
							-1, -1, -1, -1, -1, -1, -1, -1
						};
						for(int inds = 0; inds < 6; inds++) 
						{
							unsigned int cubeIndice = cubeIndices[inds + 6 * n];
							if(addedVerts[cubeIndice] == -1) 
							{
								addedVerts[cubeIndice] = vertices.size();
								vertices.push_back(cubePositions[cubeIndice] + glm::vec3(i,j,k));
								textureOffsets.push_back(glm::vec3(materialAtIndex.uv.y));
								normals.push_back(glm::vec3(neighbors[n]));
							}
							indices.push_back(addedVerts[cubeIndice]);
						}
					}
				}
				//CUBE END
				glm::vec3 vertlist[12];
				unsigned char cubeindex = 0;
				Voxel* voxelGrid[8] =
				{
					data->getVoxelWrapped(glm::ivec3(i,j,k) + gridI[0]),
					data->getVoxelWrapped(glm::ivec3(i,j,k) + gridI[1]),
					data->getVoxelWrapped(glm::ivec3(i,j,k) + gridI[2]),
					data->getVoxelWrapped(glm::ivec3(i,j,k) + gridI[3]),
					data->getVoxelWrapped(glm::ivec3(i,j,k) + gridI[4]),
					data->getVoxelWrapped(glm::ivec3(i,j,k) + gridI[5]),
					data->getVoxelWrapped(glm::ivec3(i,j,k) + gridI[6]),
					data->getVoxelWrapped(glm::ivec3(i,j,k) + gridI[7])
				};

				bool hasGrassDecoration = false;
				for(int i = 0; i < 8; i++) 
					if(VoxelMaterials::getInstance().getById(voxelGrid[i]->voxelTypeId).decorations == DecorationType::Grass) 
					{
						hasGrassDecoration = true;
						break;
					}

				float scalarGrid[8] =
				{
					float(voxelGrid[0]->amount) / 32767.0f,
					float(voxelGrid[1]->amount) / 32767.0f,
					float(voxelGrid[2]->amount) / 32767.0f,
					float(voxelGrid[3]->amount) / 32767.0f,
					float(voxelGrid[4]->amount) / 32767.0f,
					float(voxelGrid[5]->amount) / 32767.0f,
					float(voxelGrid[6]->amount) / 32767.0f,
					float(voxelGrid[7]->amount) / 32767.0f 
				};

				VoxelMaterial materialGrid[8] = {
					VoxelMaterials::getInstance().getById(voxelGrid[0]->voxelTypeId),
					VoxelMaterials::getInstance().getById(voxelGrid[1]->voxelTypeId),
					VoxelMaterials::getInstance().getById(voxelGrid[2]->voxelTypeId),
					VoxelMaterials::getInstance().getById(voxelGrid[3]->voxelTypeId),
					VoxelMaterials::getInstance().getById(voxelGrid[4]->voxelTypeId),
					VoxelMaterials::getInstance().getById(voxelGrid[5]->voxelTypeId),
					VoxelMaterials::getInstance().getById(voxelGrid[6]->voxelTypeId),
					VoxelMaterials::getInstance().getById(voxelGrid[7]->voxelTypeId) 
				};

				if(scalarGrid[7] > 0.0f) cubeindex |= (1 << 0);
				if(scalarGrid[6] > 0.0f) cubeindex |= (1 << 1);
				if(scalarGrid[5] > 0.0f) cubeindex |= (1 << 2);
				if(scalarGrid[4] > 0.0f) cubeindex |= (1 << 3);
				if(scalarGrid[3] > 0.0f) cubeindex |= (1 << 4);
				if(scalarGrid[2] > 0.0f) cubeindex |= (1 << 5);
				if(scalarGrid[1] > 0.0f) cubeindex |= (1 << 6);
				if(scalarGrid[0] > 0.0f) cubeindex |= (1 << 7);

				float toffsets[3];

				for (int l = 0; edgeTable[cubeindex][l*2] != -1; l++) {
					int edgeFrom = edgeTable[cubeindex][l*2];
					int edgeTo = edgeTable[cubeindex][l*2 + 1];
					this->indices.push_back(this->vertices.size());
					
					glm::vec3 vertPos = glm::mix(grid[edgeFrom], grid[edgeTo], EdgeIntersection(scalarGrid[edgeFrom], scalarGrid[edgeTo])) + glm::vec3(i, j, k);

					VoxelMaterial& vertMaterial = materialGrid[edgeFrom];

					toffsets[l % 3] = vertMaterial.uv.y;

					this->vertices.push_back(vertPos);
					if (l % 3 == 2)
					{
						this->textureOffsets.push_back({toffsets[0], toffsets[1], toffsets[2]});
						this->textureOffsets.push_back({toffsets[0], toffsets[1], toffsets[2]});
						this->textureOffsets.push_back({toffsets[0], toffsets[1], toffsets[2]});
						unsigned int lastv = vertices.size() - 1;
						glm::vec3 dv1 = vertices[lastv - 1] - vertices[lastv];
						glm::vec3 dv2 = vertices[lastv - 2] - vertices[lastv];
						glm::vec3 normal = normalize(cross(dv2, dv1));
						if(isnan(normal.x)) 
						{
							normal = glm::vec3(0.0f, 1.0f, 0.0f);
						}

						if(materialAtIndex.decorations == DecorationType::Grass) 
						{
							for(int grassId = 0; grassId < 1; grassId++) 
							{
								if(normal.y > 0.8f) 
								{
									//get two random random points on a triangle:
									float r1p1 = Noise::get((int)(this) + i + grassId, j + l + grassId * 4, (k + 0xfaded101) * 1);//float(rand() % 32000) / 32000.0f;
									float r2p1 = Noise::get((int)(this) + i + grassId, j + l + grassId * 3, (k + 0xc0ffebea) * 2);//float(rand() % 32000) / 32000.0f;
									float r1p2 = Noise::get((int)(this) + i + grassId, j + l + grassId * 2, (k + 0x041de05c) * 3);//float(rand() % 32000) / 32000.0f;
									float r2p2 = Noise::get((int)(this) + i + grassId, j + l + grassId * 1, (k + 0xca4471c1) * 4);//float(rand() % 32000) / 32000.0f;
									if(r1p1 + r2p1 > 1.0f) 
									{
										r1p1 = 1.0f - r1p1;
										r2p1 = 1.0f - r2p1;
									} 
									if(r1p2 + r2p2 > 1.0f) 
									{
										r1p2 = 1.0f - r1p2;
										r2p2 = 1.0f - r2p2;
									} 
									glm::vec3 p1 = vertices[lastv] + dv1 * r1p1 + dv2 * r2p1;
									glm::vec3 p2 = vertices[lastv] + dv1 * r1p2 + dv2 * r2p2;
									glm::vec3 pd = p2 - p1;
									float h = glm::length(pd);
									if(h < 0.5f) 
									{
										glm::vec3 pdu = glm::normalize(pd);
										p1 -= pdu * 0.25f;
										p2 += pdu * 0.25f;
										h += 0.5f;
										pd += pdu * 0.5f;
									} 
									glm::vec3 grassNormal = glm::vec3(p1.z - p2.z, 0.0f, p2.x - p1.x);
									int vdecbase = decVertices.size();
									decVertices.push_back(p1);
									decVertices.push_back(p2);
									decVertices.push_back(p1 + glm::vec3(0.0f, h, 0.0f));
									decVertices.push_back(p2 + glm::vec3(0.0f, h, 0.0f));
									decUVs.push_back(glm::vec2(0.0f, 0.0f));
									decUVs.push_back(glm::vec2(1.0f, 0.0f));
									decUVs.push_back(glm::vec2(0.0f, 1.0f));
									decUVs.push_back(glm::vec2(1.0f, 1.0f));
									decNormals.push_back(grassNormal);
									decNormals.push_back(grassNormal);
									decNormals.push_back(grassNormal);
									decNormals.push_back(grassNormal);
									decIndices.push_back(vdecbase + 0);
									decIndices.push_back(vdecbase + 1);
									decIndices.push_back(vdecbase + 3);
									decIndices.push_back(vdecbase + 3);
									decIndices.push_back(vdecbase + 2);
									decIndices.push_back(vdecbase + 0);


									decVertices.push_back(p1);
									decVertices.push_back(p2);
									decVertices.push_back(p1 + glm::vec3(0.0f, h, 0.0f));
									decVertices.push_back(p2 + glm::vec3(0.0f, h, 0.0f));
									decUVs.push_back(glm::vec2(0.0f, 0.0f));
									decUVs.push_back(glm::vec2(1.0f, 0.0f));
									decUVs.push_back(glm::vec2(0.0f, 1.0f));
									decUVs.push_back(glm::vec2(1.0f, 1.0f));
									decNormals.push_back(-grassNormal);
									decNormals.push_back(-grassNormal);
									decNormals.push_back(-grassNormal);
									decNormals.push_back(-grassNormal);
									decIndices.push_back(vdecbase + 0 + 4);
									decIndices.push_back(vdecbase + 2 + 4);
									decIndices.push_back(vdecbase + 3 + 4);
									decIndices.push_back(vdecbase + 3 + 4);
									decIndices.push_back(vdecbase + 1 + 4);
									decIndices.push_back(vdecbase + 0 + 4);

									glm::vec3 p1r = (p1 + p2 + glm::vec3(pd.z, 0.0f, -pd.x)) * 0.5f;
									glm::vec3 p2r = (p1 + p2 - glm::vec3(pd.z, 0.0f, -pd.x)) * 0.5f;
									p1 = p1r;
									p2 = p2r;

									decVertices.push_back(p1);
									decVertices.push_back(p2);
									decVertices.push_back(p1 + glm::vec3(0.0f, h, 0.0f));
									decVertices.push_back(p2 + glm::vec3(0.0f, h, 0.0f));
									decUVs.push_back(glm::vec2(0.0f, 0.0f));
									decUVs.push_back(glm::vec2(1.0f, 0.0f));
									decUVs.push_back(glm::vec2(0.0f, 1.0f));
									decUVs.push_back(glm::vec2(1.0f, 1.0f));
									decNormals.push_back(-glm::vec3(grassNormal.z, 0.0f, -grassNormal.x));
									decNormals.push_back(-glm::vec3(grassNormal.z, 0.0f, -grassNormal.x));
									decNormals.push_back(-glm::vec3(grassNormal.z, 0.0f, -grassNormal.x));
									decNormals.push_back(-glm::vec3(grassNormal.z, 0.0f, -grassNormal.x));
									decIndices.push_back(vdecbase + 0 + 8);
									decIndices.push_back(vdecbase + 1 + 8);
									decIndices.push_back(vdecbase + 3 + 8);
									decIndices.push_back(vdecbase + 3 + 8);
									decIndices.push_back(vdecbase + 2 + 8);
									decIndices.push_back(vdecbase + 0 + 8);


									decVertices.push_back(p1);
									decVertices.push_back(p2);
									decVertices.push_back(p1 + glm::vec3(0.0f, h, 0.0f));
									decVertices.push_back(p2 + glm::vec3(0.0f, h, 0.0f));
									decUVs.push_back(glm::vec2(0.0f, 0.0f));
									decUVs.push_back(glm::vec2(1.0f, 0.0f));
									decUVs.push_back(glm::vec2(0.0f, 1.0f));
									decUVs.push_back(glm::vec2(1.0f, 1.0f));
									decNormals.push_back(glm::vec3(grassNormal.z, 0.0f, -grassNormal.x));
									decNormals.push_back(glm::vec3(grassNormal.z, 0.0f, -grassNormal.x));
									decNormals.push_back(glm::vec3(grassNormal.z, 0.0f, -grassNormal.x));
									decNormals.push_back(glm::vec3(grassNormal.z, 0.0f, -grassNormal.x));
									decIndices.push_back(vdecbase + 0 + 12);
									decIndices.push_back(vdecbase + 2 + 12);
									decIndices.push_back(vdecbase + 3 + 12);
									decIndices.push_back(vdecbase + 3 + 12);
									decIndices.push_back(vdecbase + 1 + 12);
									decIndices.push_back(vdecbase + 0 + 12);
								}
							}
						}
						
						if(materialAtIndex.smooth == false) 
						{
							normals.push_back(normal);
							normals.push_back(normal);
							normals.push_back(normal);
						}
						else 
						{
							//const float normalSampleBias = 0.5f;
							//glm::vec3 smoothNormal = data->getNormal(vertPos);
							/*glm::normalize(vec3(
								data->getInterpolatedFieldValue(vertPos + vec3(0.005f) + vec3(-normalSampleBias, 0.0f, 0.0f)) - data->getInterpolatedFieldValue(vertPos + vec3(0.005f) + vec3(normalSampleBias, 0.0f, 0.0f)),
								data->getInterpolatedFieldValue(vertPos + vec3(0.005f) + vec3(0.0f, -normalSampleBias, 0.0f)) - data->getInterpolatedFieldValue(vertPos + vec3(0.005f) + vec3(0.0f, normalSampleBias, 0.0f)),
								data->getInterpolatedFieldValue(vertPos + vec3(0.005f) + vec3(0.0f, 0.0f, -normalSampleBias)) - data->getInterpolatedFieldValue(vertPos + vec3(0.005f) + vec3(0.0f, 0.0f, normalSampleBias))
							));*/ 
							this->normals.push_back(data->getNormal(vertices[lastv - 2]));
							this->normals.push_back(data->getNormal(vertices[lastv - 1]));
							this->normals.push_back(data->getNormal(vertices[lastv - 0]));
						}
					}
				}
			}
		}
	}
	this->buildDataFromVectors();
	decorations.buildDataFromVectors(decVertices, decUVs, decNormals, decIndices);
	hasDecorations = decVertices.size() > 0;
}

void MarchingCubesVoxelMesh::uploadMesh() 
{
	if(vertices.size() > 0) this->uploadData();
	if(this->decorations.getMesh().indexCount > 0) decorations.uploadData(GL_DYNAMIC_DRAW);
}

void MarchingCubesVoxelMesh::buildDataFromVectors() 
{
	meshData.clear();
	this->mesh.boundingBox.minX = 100000;
	this->mesh.boundingBox.minY = 100000;
	this->mesh.boundingBox.minZ = 100000;
	this->mesh.boundingBox.maxX = -100000;
	this->mesh.boundingBox.maxY = -100000;
	this->mesh.boundingBox.maxZ = -100000;
	for(int i = 0; i < vertices.size(); i++) 
	{
		meshData.push_back(vertices[i].x);
		meshData.push_back(vertices[i].y);
		meshData.push_back(vertices[i].z);
		meshData.push_back(normals[i].x);
		meshData.push_back(normals[i].y);
		meshData.push_back(normals[i].z);
		meshData.push_back(textureOffsets[i].x);
		meshData.push_back(textureOffsets[i].y);
		meshData.push_back(textureOffsets[i].z);
		this->mesh.boundingBox.minX = glm::min(this->mesh.boundingBox.minX, vertices[i].x);
		this->mesh.boundingBox.minY = glm::min(this->mesh.boundingBox.minX, vertices[i].y);
		this->mesh.boundingBox.minZ = glm::min(this->mesh.boundingBox.minX, vertices[i].z);
		this->mesh.boundingBox.maxX = glm::max(this->mesh.boundingBox.maxX, vertices[i].x);
		this->mesh.boundingBox.maxY = glm::max(this->mesh.boundingBox.maxY, vertices[i].y);
		this->mesh.boundingBox.maxZ = glm::max(this->mesh.boundingBox.maxZ, vertices[i].z);
	}
	mesh.indexCount = indices.size();
	mesh.indexOffset = 0;
	mesh.vertexOffset = 0;
}

void MarchingCubesVoxelMesh::uploadData() 
{
	if(vao == 0) 
	{
		vao.create();
	}
	glBindVertexArray(vao);
	glGenBuffers(1, &vbo);
	glGenBuffers(1, &ebo);
	glBindBuffer(GL_ARRAY_BUFFER, vbo);
	glBufferData(GL_ARRAY_BUFFER, sizeof(float) * meshData.size(), meshData.data(), GL_DYNAMIC_DRAW);
	glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, ebo);
	glBufferData(GL_ELEMENT_ARRAY_BUFFER, sizeof(unsigned int) * indices.size(), indices.data(), GL_DYNAMIC_DRAW);
	vao.vertexAttribPointer(Kirino::VertexArrayObject::VertexAttribPointer(true, vbo, 0, 3, GL_FLOAT, false, sizeof(float) * (9), 0, 0));
	vao.vertexAttribPointer(Kirino::VertexArrayObject::VertexAttribPointer(true, vbo, 1, 3, GL_FLOAT, false, sizeof(float) * (9), (void*)(sizeof(float) * 3), 0));
	vao.vertexAttribPointer(Kirino::VertexArrayObject::VertexAttribPointer(true, vbo, 2, 3, GL_FLOAT, false, sizeof(float) * (9), (void*)(sizeof(float) * 6), 0));
	vao.updateAttribs();	
	mesh.vao = &vao;
}

void MarchingCubesVoxelMesh::clear() 
{
	this->vertices.clear();
	this->normals.clear();
	this->textureOffsets.clear();
	this->indices.clear();
	decVertices.clear();
	decNormals.clear();
	decUVs.clear();
	decIndices.clear();
}
